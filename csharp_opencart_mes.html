public class OrderRepository
{
    private readonly string _connectionString = "Server=127.0.0.1;Port=3306;Database=db_opencart;Uid=root;Pwd=;CharSet=utf8mb4;";

    // 1. 查詢未完工訂單
    public async Task<IEnumerable<OrderModel>> GetPendingOrdersAsync()
    {
        using var conn = new MySqlConnection(_connectionString);
        string sql = @"
            SELECT 
                o.order_id AS OrderId, 
                o.firstname AS FirstName, 
                o.lastname AS LastName, 
                o.total AS Total, 
                o.order_status_id AS OrderStatusId, 
                o.date_added AS DateAdded,
                p.name AS ProductName,
                p.quantity AS Quantity,
                opt.value AS OptionValue
            FROM oc_order o
            LEFT JOIN oc_order_product p ON o.order_id = p.order_id
            LEFT JOIN oc_order_option opt ON p.order_product_id = opt.order_product_id
            WHERE o.order_status_id != '5' AND o.order_status_id != '7';";
        return await conn.QueryAsync<OrderModel>(sql);
    }
}

步驟四：實作現場一鍵緊急加急 (Emergency Push) 雙表同步

使用 MySQL Transaction 確保更新 oc_order 狀態為 99 與寫入 oc_order_history 歷程之原子性 (Atomicity)：

Repositories/OrderRepository.cs (加急方法) 
public async Task<bool> PushUrgentOrderAsync(int orderId)
{
    using var conn = new MySqlConnection(_connectionString);
    await conn.OpenAsync();
    using var trans = await conn.BeginTransactionAsync();
    try
    {
        string updateSql = "UPDATE oc_order SET order_status_id = 99 WHERE order_id = @OrderId;";
        await conn.ExecuteAsync(updateSql, new { OrderId = orderId }, trans);

        string historySql = @"
            INSERT INTO oc_order_history (order_id, order_status_id, notify, comment, date_added)
            VALUES (@OrderId, 99, 1, '現場作業員透過 C# MES 上位機啟動加急作業', NOW());";
        await conn.ExecuteAsync(historySql, new { OrderId = orderId }, trans);

        await trans.CommitAsync();
        return true;
    }
    catch
    {
        await trans.RollbackAsync();
        return false;
    }
}

public async Task StartMachiningAsync(int orderId, IProgress<int> progress)
{
    // 1. 切換狀態為加工中 (黃燈)
    MachineStatus = Status.Machining;
    
    // 2. 模擬 4 秒切削進度
    for (int i = 0; i <= 100; i += 5)

        await Task.Delay(200);
        progress.Report(i);

    MachineStatus = Status.Complete; // 完工藍燈
}

public async Task RefreshKpiAsync()
{
    using var conn = new MySqlConnection(_connectionString);
    
    // 待生產工件兩階段巢狀子查詢
    string pendingSql = @"
        SELECT SUM(total_quantity) AS grand_total_quantity FROM (
            SELECT oc_order.order_id, oc_order.order_status_id, SUM(oc_order_product.quantity) AS total_quantity 
            FROM oc_order LEFT JOIN oc_order_product ON oc_order.order_id = oc_order_product.order_id 
            WHERE oc_order.order_status_id < '5' 
            GROUP BY oc_order.order_id, oc_order.order_status_id
        ) AS subquery;";
    PendingTotal = await conn.ExecuteScalarAsync<int>(pendingSql);

    // 已完工合格入庫兩階段巢狀子查詢
    string completeSql = @"
        SELECT SUM(total_quantity) AS grand_total_quantity FROM (
            SELECT oc_order.order_id, oc_order.order_status_id, SUM(oc_order_product.quantity) AS total_quantity 
            FROM oc_order LEFT JOIN oc_order_product ON oc_order.order_id = oc_order_product.order_id 
            WHERE oc_order.order_status_id = '5' 
            GROUP BY oc_order.order_id, oc_order.order_status_id
        ) AS subquery;";
    CompleteTotal = await conn.ExecuteScalarAsync<int>(completeSql);

    // 計算良率
    int totalProduced = CompleteTotal + Defects;
    YieldRate = totalProduced > 0 ? ((double)CompleteTotal / totalProduced) * 100.0 : 100.0;
}

        // 現場加急調度 (Emergency Push)
        private async void BtnUrgent_Click(object sender, RoutedEventArgs e)
        {
            using var conn = new MySqlConnection(ConnStr);
            await conn.OpenAsync();
            using var trans = await conn.BeginTransactionAsync();
            try
            {
                await conn.ExecuteAsync("UPDATE oc_order SET order_status_id = 99 WHERE order_id = @Id;", new { Id = _selectedOrderId }, trans);
                await conn.ExecuteAsync(@"INSERT INTO oc_order_history (order_id, order_status_id, notify, comment, date_added)
                                          VALUES (@Id, 99, 1, 'C# 上位機觸發加急排產', NOW());", new { Id = _selectedOrderId }, trans);
                await trans.CommitAsync();

                // 更新 LED 燈號為加急橙燈
                LedEllipse.Fill = new SolidColorBrush(Color.FromRgb(249, 115, 22));
                MessageBox.Show($"工單 #{_selectedOrderId} 加急調度成功！狀態已升級至 99！", "MES 系統提示");
                await LoadDataAsync();
            }
            catch (Exception ex)
            {
                await trans.RollbackAsync();
                MessageBox.Show("加急失敗: " + ex.Message);
            }
        }

        // 兩階段巢狀子查詢與良率計算
        private async Task RefreshKpiAsync()
        {
            using var conn = new MySqlConnection(ConnStr);
            string pendingSql = @"SELECT SUM(total_quantity) FROM (
                                    SELECT SUM(quantity) AS total_quantity FROM oc_order o 
                                    LEFT JOIN oc_order_product p ON o.order_id = p.order_id 
                                    WHERE o.order_status_id < '5' GROUP BY o.order_id) t;";
            int pending = await conn.ExecuteScalarAsync<int>(pendingSql);

            string completeSql = @"SELECT SUM(total_quantity) FROM (
                                     SELECT SUM(quantity) AS total_quantity FROM oc_order o 
                                     LEFT JOIN oc_order_product p ON o.order_id = p.order_id 
                                     WHERE o.order_status_id = '5' GROUP BY o.order_id) t;";
            int complete = await conn.ExecuteScalarAsync<int>(completeSql);

            int total = complete + _defects;
            double yieldRate = total > 0 ? ((double)complete / total) * 100.0 : 100.0;

            TxtPending.Text = pending + " 件";
            TxtComplete.Text = complete + " 件";
            TxtYield.Text = yieldRate.ToString("F1") + "%";
        }
    }

步驟四：實作 Bearer Token 自動認證處理器 (RestAdminApiClient.cs)

撰寫具備 Token 快取與過期自動更新機制的 REST API Client：

RestAdminApiClient.cs 
using System;
using System.Threading.Tasks;
using RestSharp;
using RestSharp.Authenticators;

public class RestAdminApiClient
{
    private readonly RestClient _client;
    private readonly string _clientId;
    private readonly string _clientSecret;
    private string? _accessToken;
    private DateTime _tokenExpiry = DateTime.MinValue;

    public RestAdminApiClient(string baseUrl, string clientId, string clientSecret)
    {
        _clientId = clientId;
        _clientSecret = clientSecret;
        _client = new RestClient(baseUrl);
    }

    // 確保 Token 在有效期限內 (未過期則重用，過期自動重簽)
    private async Task EnsureAuthenticatedAsync()
    {
        if (!string.IsNullOrEmpty(_accessToken) && DateTime.UtcNow < _tokenExpiry)
            return;

        var request = new RestRequest("oauth/token", Method.Post);
        request.AddJsonBody(new
        {
            client_id = _clientId,
            client_secret = _clientSecret,
            grant_type = "client_credentials"
        });

        var response = await _client.ExecuteAsync<TokenResponse>(request);
        if (response.IsSuccessful && response.Data != null)
        {
            _accessToken = response.Data.access_token;
            _tokenExpiry = DateTime.UtcNow.AddSeconds(response.Data.expires_in - 60);
            _client.Authenticator = new JwtAuthenticator(_accessToken);
            Console.WriteLine("[REST] 成功取得 Bearer Token！");
        }
        else
        {
            throw new Exception($"[REST] OAuth2 授權失敗: {response.ErrorMessage}");
        }
    }

    public record TokenResponse(string access_token, int expires_in);
}

步驟五：呼叫 GET /orders 查詢待產工單清單

利用 RESTful 查詢參數過濾未完成工單（filter_order_status_id=1）：

工單查詢方法 
    // 查詢待生產工單列表
    public async Task<string> GetPendingOrdersAsync()
    {
        await EnsureAuthenticatedAsync();

        var request = new RestRequest("orders", Method.Get);
        request.AddQueryParameter("filter_order_status_id", "1"); // 待處理 (Pending)
        request.AddQueryParameter("limit", "50");

        var response = await _client.ExecuteAsync(request);
        Console.WriteLine($"[REST] 查詢工單成功: {response.Content}");
        return response.Content ?? "{}";
    }

步驟六：呼叫 PUT /orders/{id}/status 執行機台完工回寫

使用標準 PUT 動詞將工單狀態修改為已完成 (5)，並寫入品質歷程備註：

PUT 更新狀態方法 
    // 機台完工回寫：更新狀態碼為 5 (Complete)
    public async Task<bool> CompleteOrderAsync(int orderId, string comment)
    {
        await EnsureAuthenticatedAsync();

        var request = new RestRequest($"orders/{orderId}/status", Method.Put);
        request.AddJsonBody(new
        {
            status_id = 5,
            comment = comment,
            notify = true
        });

        var response = await _client.ExecuteAsync(request);
        Console.WriteLine($"[REST] 完工回寫結果: {response.StatusCode}");
        return response.IsSuccessful;
    }
